// GenClass regression test for finding f2 (race), generated from run r21.

import { test, expect } from "@playwright/test";

const APP = process.env.GENCLASS_APP_URL ?? "http://localhost:8080";
const SCHEDULE = {"order":"seeded","seed":3,"delay":{"minMs":0,"maxMs":900},"maxHoldMs":2400};

test("/: the final state does not depend on response order (GenClass f2)", async ({ page }) => {
  await page.addInitScript((s) => { (window as any).__GENCLASS_SCHEDULE__ = s; }, SCHEDULE);
  await page.goto(APP + "/#/");
  expect(await page.evaluate(() => !!(window as any).__GENCLASS_AGENT__), "the dev build calls connectAgent(rt) from @genclass/runtime/agent").toBe(true);
  await page.evaluate(() => (window as any).__GENCLASS_AGENT__.whenSettled(15000));

  const snap = await page.evaluate(() => (window as any).__GENCLASS_AGENT__.snapshot());
  void witnesses; void errors; void snap;
  expect(snap.text, "the page shows what it shows with responses in request order").toContain("8m ago");
  expect(snap.text, "the page does not show what only the failing order showed").not.toContain("9m ago");
